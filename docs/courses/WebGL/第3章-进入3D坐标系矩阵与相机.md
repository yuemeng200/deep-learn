# 第 3 章：进入 3D——坐标系、矩阵与相机

## 回扣本质

上一章我们画出了二维三角形。但"3D"的本质是：物体存在于一个三维空间，通过一套坐标变换，最终投影到二维屏幕上。这一章我们打通这套变换链——它是所有 3D 图形（无论 WebGL、Three.js 还是 Unity）背后共同的数学基础。

---

## 一、五大坐标空间

```
局部空间 → 世界空间 → 观察空间 → 裁剪空间 → 屏幕空间
(Local)    (World)    (View)      (Clip)       (Screen)
    M矩阵 ↗      V矩阵 ↗      P矩阵 ↗      GPU自动完成
```

**局部空间**：以物体自身为原点。一个立方体的顶点是 ±1 范围内的坐标，不管它在世界哪个位置。

**世界空间**：将物体放入世界坐标系，用 Model 矩阵（M）变换。

**观察空间**：以相机为原点重新描述世界，用 View 矩阵（V）变换。

**裁剪空间**：施加透视效果，用 Projection 矩阵（P）变换。这是 `gl_Position` 的输出空间。

**顶点着色器的核心任务**：
```glsl
gl_Position = u_projection * u_view * u_model * a_position;
```

---

## 二、矩阵变换的直觉

一个 4×4 矩阵就是一种空间变换的"配方"：
- **平移矩阵**：把物体移到 (x, y, z)
- **旋转矩阵**：绕某个轴旋转 θ 角
- **缩放矩阵**：把物体缩放 (sx, sy, sz) 倍
- **透视投影矩阵**：模拟相机焦距，产生近大远小效果

多个变换通过矩阵乘法合并成一个矩阵——GPU 一次乘法完成所有变换。

## 三、旋转的 3D 立方体

```html
<!DOCTYPE html>
<html>
<head>
  <style>
    body { margin: 0; background: #111; display: flex; justify-content: center; align-items: center; height: 100vh; }
  </style>
</head>
<body>
<canvas id="c" width="600" height="400"></canvas>
<script>
const canvas = document.getElementById('c');
const gl = canvas.getContext('webgl');
gl.enable(gl.DEPTH_TEST);

const vertSrc = `
  attribute vec4 a_position;
  attribute vec4 a_color;
  uniform mat4 u_mvp;
  varying vec4 v_color;
  void main() {
    gl_Position = u_mvp * a_position;
    v_color = a_color;
  }
`;
const fragSrc = `
  precision mediump float;
  varying vec4 v_color;
  void main() {
    gl_FragColor = v_color;
  }
`;

function makeShader(type, src) {
  const s = gl.createShader(type);
  gl.shaderSource(s, src);
  gl.compileShader(s);
  return s;
}
const prog = gl.createProgram();
gl.attachShader(prog, makeShader(gl.VERTEX_SHADER, vertSrc));
gl.attachShader(prog, makeShader(gl.FRAGMENT_SHADER, fragSrc));
gl.linkProgram(prog);
gl.useProgram(prog);

const v = (x,y,z,r,g,b) => [x,y,z,r,g,b,1];
const faces = [
  ...v(-1,-1, 1, 0.9,0.2,0.2), ...v( 1,-1, 1, 0.9,0.2,0.2),
  ...v( 1, 1, 1, 0.9,0.2,0.2), ...v(-1, 1, 1, 0.9,0.2,0.2),
  ...v( 1,-1,-1, 0.2,0.4,0.9), ...v(-1,-1,-1, 0.2,0.4,0.9),
  ...v(-1, 1,-1, 0.2,0.4,0.9), ...v( 1, 1,-1, 0.2,0.4,0.9),
  ...v(-1, 1, 1, 0.2,0.8,0.3), ...v( 1, 1, 1, 0.2,0.8,0.3),
  ...v( 1, 1,-1, 0.2,0.8,0.3), ...v(-1, 1,-1, 0.2,0.8,0.3),
  ...v(-1,-1,-1, 0.9,0.8,0.1), ...v( 1,-1,-1, 0.9,0.8,0.1),
  ...v( 1,-1, 1, 0.9,0.8,0.1), ...v(-1,-1, 1, 0.9,0.8,0.1),
  ...v( 1,-1, 1, 0.7,0.2,0.9), ...v( 1,-1,-1, 0.7,0.2,0.9),
  ...v( 1, 1,-1, 0.7,0.2,0.9), ...v( 1, 1, 1, 0.7,0.2,0.9),
  ...v(-1,-1,-1, 0.9,0.5,0.1), ...v(-1,-1, 1, 0.9,0.5,0.1),
  ...v(-1, 1, 1, 0.9,0.5,0.1), ...v(-1, 1,-1, 0.9,0.5,0.1),
];

const vtxBuf = gl.createBuffer();
gl.bindBuffer(gl.ARRAY_BUFFER, vtxBuf);
gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(faces), gl.STATIC_DRAW);

const indices = [];
for (let i = 0; i < 6; i++) {
  const b = i * 4;
  indices.push(b,b+1,b+2, b,b+2,b+3);
}
const idxBuf = gl.createBuffer();
gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, idxBuf);
gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, new Uint16Array(indices), gl.STATIC_DRAW);

const stride = 7 * 4;
const posLoc = gl.getAttribLocation(prog, 'a_position');
gl.enableVertexAttribArray(posLoc);
gl.vertexAttribPointer(posLoc, 3, gl.FLOAT, false, stride, 0);

const colLoc = gl.getAttribLocation(prog, 'a_color');
gl.enableVertexAttribArray(colLoc);
gl.vertexAttribPointer(colLoc, 4, gl.FLOAT, false, stride, 3 * 4);

function multiply(a, b) {
  const r = new Float32Array(16);
  for (let i = 0; i < 4; i++)
    for (let j = 0; j < 4; j++)
      for (let k = 0; k < 4; k++)
        r[j*4+i] += a[k*4+i] * b[j*4+k];
  return r;
}

function perspective(fovY, aspect, near, far) {
  const f = 1.0 / Math.tan(fovY / 2);
  const d = near - far;
  return new Float32Array([
    f/aspect, 0, 0, 0,
    0, f, 0, 0,
    0, 0, (far+near)/d, -1,
    0, 0, (2*far*near)/d, 0,
  ]);
}

function lookAt(ex,ey,ez, cx,cy,cz) {
  let zx=ex-cx, zy=ey-cy, zz=ez-cz;
  let len=Math.sqrt(zx*zx+zy*zy+zz*zz);
  zx/=len; zy/=len; zz/=len;
  let xx= zy, xy=-zx, xz=0;
  len=Math.sqrt(xx*xx+xy*xy+xz*xz);
  xx/=len; xy/=len; xz/=len;
  const yx=zy*xz-zz*xy, yy=zz*xx-zx*xz, yz=zx*xy-zy*xx;
  return new Float32Array([
     xx, yx, zx, 0,
     xy, yy, zy, 0,
     xz, yz, zz, 0,
    -(xx*ex+xy*ey+xz*ez), -(yx*ex+yy*ey+yz*ez), -(zx*ex+zy*ey+zz*ez), 1
  ]);
}

function rotateY(t) {
  const c=Math.cos(t), s=Math.sin(t);
  return new Float32Array([c,0,-s,0, 0,1,0,0, s,0,c,0, 0,0,0,1]);
}
function rotateX(t) {
  const c=Math.cos(t), s=Math.sin(t);
  return new Float32Array([1,0,0,0, 0,c,s,0, 0,-s,c,0, 0,0,0,1]);
}

const mvpLoc = gl.getUniformLocation(prog, 'u_mvp');
const P = perspective(Math.PI/4, canvas.width/canvas.height, 0.1, 100);
const V = lookAt(0, 1.5, 5,  0, 0, 0);

let t = 0;
function render() {
  t += 0.01;
  gl.viewport(0, 0, canvas.width, canvas.height);
  gl.clearColor(0.1, 0.1, 0.15, 1);
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

  const M = multiply(rotateY(t), rotateX(t * 0.5));
  const mvp = multiply(P, multiply(V, M));
  gl.uniformMatrix4fv(mvpLoc, false, mvp);

  gl.drawElements(gl.TRIANGLES, 36, gl.UNSIGNED_SHORT, 0);
  requestAnimationFrame(render);
}
render();
</script>
</body>
</html>
```

## 四、关键决策解析

**为什么要 `gl.enable(gl.DEPTH_TEST)`？**
没有深度测试，WebGL 按绘制顺序覆盖像素。开启后，GPU 用深度缓冲区自动保留最近的面。

**索引缓冲区的意义**
立方体 6 面，每面 4 顶点共 24 个，用索引后顶点数据只存一次，通过整数索引引用，节省显存。

**`uniformMatrix4fv` 的 `false` 是什么？**
表示不转置矩阵。WebGL 使用列主序，我们的矩阵已按列主序写好。

## 章末思考题

**题 1（理解）**：MVP 矩阵的顺序是 `P * V * M`，能不能改成 `M * V * P`？为什么？

> **参考答案**：不能。矩阵乘法不满足交换律。含义是：先局部→世界（M），再世界→相机（V），最后施加投影（P）。可以这样记：离顶点最近的矩阵先作用。

**题 2（延伸）**：透视投影和正交投影有什么区别？各自适合什么场景？

> **参考答案**：透视投影产生近大远小效果，适合 3D 游戏、场景漫游。正交投影平行线保持平行，适合 2D 游戏、CAD 工程图、等距视图。Three.js 的 `PerspectiveCamera` 和 `OrthographicCamera` 对应这两种。

**题 3（挑战）**：如果想让相机绕立方体旋转一圈，`lookAt` 的参数应该怎么随时间变化？

> **参考答案**：`lookAt(Math.sin(t)*5, 1.5, Math.cos(t)*5, 0, 0, 0)`。相机沿 XZ 平面的圆轨迹运动，X = radius × sin(t)，Z = radius × cos(t)，始终看向原点。这是"环绕相机"的基本实现方式。
