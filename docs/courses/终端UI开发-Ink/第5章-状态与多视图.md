  view: 'list',
  tasks: [],

  activeFilter: 'all',
  nextId: 1,
};

function reducer(state, action) {
  switch (action.type) {
    case 'SET_FILTER':
      return {
        ...state,
        activeFilter: action.payload,
        selectedIndex: 0,   // 联动重置，防止 index 越界
      };
    case 'NAVIGATE':
      return {
        ...state,
        selectedIndex: Math.max(0,
          Math.min(action.max, state.selectedIndex + action.delta)
        ),
      };
    // ...
  }
}

## 实战代码：完整多视图 taskr

```jsx
import React, { useReducer } from 'react';
import { render, Text, Box, useInput, useApp } from 'ink';
import TextInput from 'ink-text-input';

function getFiltered(tasks, filter) {
  if (filter === 'todo') return tasks.filter(t => !t.done);
  if (filter === 'done') return tasks.filter(t => t.done);
  return tasks;
}

const initialState = {
  view: 'list',
  tasks: [
    { id: 1, title: '买咖啡豆', done: false },
    { id: 2, title: '写周报',   done: true  },
    { id: 3, title: '回复邮件', done: false },
  ],
  selectedIndex: 0,
  activeFilter: 'all',
  nextId: 4,
};

function reducer(state, action) {
  const filtered = getFiltered(state.tasks, state.activeFilter);

  switch (action.type) {
    case 'SET_VIEW':
      return { ...state, view: action.payload };
    case 'SET_FILTER':
      return { ...state, activeFilter: action.payload, selectedIndex: 0 };
    case 'NAVIGATE':
      return {
        ...state,
        selectedIndex: Math.max(0,
          Math.min(filtered.length - 1, state.selectedIndex + action.delta)
        ),
      };
    case 'TOGGLE_DONE': {
      const task = filtered[state.selectedIndex];
      if (!task) return state;
      return {
        ...state,
        tasks: state.tasks.map(t =>
          t.id === task.id ? { ...t, done: !t.done } : t
        ),
      };
    }
    case 'ADD_TASK':
      if (!action.payload.trim()) return { ...state, view: 'list' };
      return {
        ...state,
        view: 'list',
        tasks: [...state.tasks, {
          id: state.nextId,
          title: action.payload.trim(),
          done: false,
        }],
        nextId: state.nextId + 1,
      };
    case 'DELETE_TASK': {
      const task = filtered[state.selectedIndex];
      if (!task) return state;
      return {
        ...state,
        tasks: state.tasks.filter(t => t.id !== task.id),
        selectedIndex: Math.max(0, state.selectedIndex - 1),
      };
    }
    default:
      return state;
  }
}

function CreateView({ onSubmit, onCancel }) {
  const [value, setValue] = React.useState('');

  useInput((input, key) => {
    if (key.escape) onCancel();
  });

  return (
    <Box flexDirection="column" padding={1} gap={1}>
      <Text color="cyan" bold>新建任务</Text>
      <Box gap={1}>
        <Text color="gray">标题：</Text>
        <TextInput
          value={value}
          onChange={setValue}
          onSubmit={onSubmit}
          placeholder="输入任务名称，Enter 确认，Esc 取消"
        />
      </Box>
    </Box>
  );
}

function App() {
  const { exit } = useApp();
  const [state, dispatch] = useReducer(reducer, initialState);

  return (
    <Box flexDirection="column">
      <Box borderStyle="round" borderColor="cyan" paddingX={1}>
        <Text color="cyan" bold>taskr  </Text>
        <Text color="gray">
          {state.view === 'list'
            ? '↑↓ 导航  空格 切换  a 新建  d 删除  q 退出'
            : '新建任务中...'}
        </Text>
      </Box>

      {state.view === 'list' && (
        // ListView 组件（参考第 4 章，dispatch 替换 setState）
        <Box>{/* ... */}</Box>
      )}
      {state.view === 'create' && (
        <CreateView
          onSubmit={title => dispatch({ type: 'ADD_TASK', payload: title })}
          onCancel={() => dispatch({ type: 'SET_VIEW', payload: 'list' })}
        />
      )}

      <Box borderStyle="round" borderColor="gray" paddingX={1}>
        <Text color="gray">
          共 {getFiltered(state.tasks, state.activeFilter).length} 条任务
        </Text>
      </Box>
    </Box>
  );
}

render(<App />);
```

## 状态设计原则

| 类型 | 放在哪里 | 例子 |
|------|---------|------|
| 多组件共享 / 联动 | 全局 reducer | tasks、view、selectedIndex |
| 只影响自身的临时 UI | 局部 useState | 输入框当前值 |
